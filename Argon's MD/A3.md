## 2. The Floating Pill Overlay (`src/components/FloatingHud.tsx`)

Because the `hud` window is configured in `tauri.conf.json` as `220x48`, frameless, transparent, and non-activating (`WS_EX_NOACTIVATE`), this component renders a sleek floating pill that listens to Tauri cross-window events (`hud-state`) to switch between **Listening** (animated waveform) and **Polishing** (Groq Llama cleanup).

```tsx
import React, { useEffect, useState } from "react";
import { listen } from "@tauri-apps/api/event";
import { Sparkles, Mic } from "lucide-react";

type HudMode = "recording" | "processing" | "error";

export default function FloatingHud() {
  const [mode, setMode] = useState<HudMode>("recording");
  const [seconds, setSeconds] = useState<number>(0);

  useEffect(() => {
    // Listen for state changes emitted by the Dashboard/Shortcut manager
    const unlistenPromise = listen<{ mode: HudMode }>("hud-state", (event) => {
      setMode(event.payload.mode);
      if (event.payload.mode === "recording") {
        setSeconds(0);
      }
    });

    return () => {
      unlistenPromise.then((unlisten) => unlisten());
    };
  }, []);

  useEffect(() => {
    if (mode !== "recording") return;
    const timer = setInterval(() => {
      setSeconds((s) => s + 0.1);
    }, 100);
    return () => clearInterval(timer);
  }, [mode]);

  return (
    <div className="w-screen h-screen bg-transparent flex items-center justify-center select-none overflow-hidden">
      <div
        className={`flex items-center justify-between px-4 py-2 rounded-full backdrop-blur-md shadow-2xl border transition-all duration-200 w-[210px] h-[42px] ${
          mode === "error"
            ? "bg-red-950/90 border-red-500/60 text-red-200"
            : mode === "processing"
            ? "bg-zinc-900/95 border-amber-500/40 text-amber-200"
            : "bg-zinc-900/95 border-emerald-500/40 text-zinc-100"
        }`}
      >
        {mode === "recording" && (
          <>
            <div className="flex items-center gap-2">
              <span className="relative flex h-2.5 w-2.5">
                <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-emerald-400 opacity-75"></span>
                <span className="relative inline-flex rounded-full h-2.5 w-2.5 bg-emerald-500"></span>
              </span>
              <Mic className="w-3.5 h-3.5 text-emerald-400" />
              <span className="text-xs font-medium tracking-wide">Listening</span>
            </div>

            {/* Animated 5-Bar CSS Waveform */}
            <div className="flex items-center gap-1 h-4">
              {[0.4, 0.9, 0.6, 1.0, 0.5].map((scale, idx) => (
                <span
                  key={idx}
                  className="w-1 bg-emerald-400 rounded-full animate-pulse"
                  style={{
                    height: `${Math.max(25, ((seconds * 10 + idx * 3) % 10) * 10 * scale)}%`,
                    animationDelay: `${idx * 120}ms`,
                  }}
                />
              ))}
            </div>

            <span className="text-[11px] font-mono text-zinc-400">
              {seconds.toFixed(1)}s
            </span>
          </>
        )}

        {mode === "processing" && (
          <div className="flex items-center justify-center w-full gap-2">
            <Sparkles className="w-4 h-4 text-amber-400 animate-spin" />
            <span className="text-xs font-medium text-amber-300">
              Groq Polishing...
            </span>
          </div>
        )}

        {mode === "error" && (
          <div className="flex items-center justify-center w-full">
            <span className="text-xs font-medium text-red-300">
              Dictation Failed
            </span>
          </div>
        )}
      </div>
    </div>
  );
}
```

## 3. Main History, Insights & Settings Dashboard (`src/components/Dashboard.tsx`)

This component manages:
1. **Push-to-Talk Global Shortcut Registration** (`CommandOrControl+Shift+Space` by default, or hold-to-talk via `@tauri-apps/plugin-global-shortcut`).
2. **Searchable Dictation History** pulled from `%APPDATA%\com.suki.windows\suki_history.db` with 1-click copy and app context badges.
3. **AI Daily Digest Generator** that summarizes your spoken work across apps.
4. **BYOK Settings Tab** to configure your Groq API key and custom shortcut.

```tsx
import React, { useEffect, useState, useRef } from "react";
import { invoke } from "@tauri-apps/api/core";
import { emit } from "@tauri-apps/api/event";
import {
  register,
  unregisterAll,
  ShortcutEvent,
} from "@tauri-apps/plugin-global-shortcut";
import {
  Mic,
  History,
  Sparkles,
  Settings,
  Copy,
  Check,
  Terminal,
  Clock,
  Zap,
  KeyRound,
} from "lucide-react";

export interface DictationEntry {
  id: number;
  timestamp: string;
  raw_text: string;
  polished_text: string;
  active_app: string;
  duration_sec: number;
  word_count: number;
}

export default function Dashboard() {
  const [activeTab, setActiveTab] = useState<"history" | "insights" | "settings">("history");
  const [entries, setEntries] = useState<DictationEntry[]>([]);
  const [apiKey, setApiKey] = useState<string>(
    () => localStorage.getItem("suki_groq_api_key") || ""
  );
  const [hotkey, setHotkey] = useState<string>(
    () => localStorage.getItem("suki_hotkey") || "CommandOrControl+Shift+Space"
  );
  const [isRecording, setIsRecording] = useState<boolean>(false);
  const [copiedId, setCopiedId] = useState<number | null>(null);
  const [digestMarkdown, setDigestMarkdown] = useState<string>("");
  const [isGeneratingDigest, setIsGeneratingDigest] = useState<boolean>(false);
  const [statusBanner, setStatusBanner] = useState<string | null>(null);

  // Guard ref so holding a key down doesn't re-trigger start_recording on key-repeat ticks
  const recordingRef = useRef<boolean>(false);
  const apiKeyRef = useRef<string>(apiKey);
  apiKeyRef.current = apiKey;

  const fetchHistory = async () => {
    try {
      const data = await invoke<DictationEntry[]>("get_recent_dictations");
      setEntries(data);
    } catch (err) {
      console.error("Failed to load history:", err);
    }
  };

  useEffect(() => {
    fetchHistory();
  }, []);

  // Register Push-to-Talk Global Hotkey via Tauri v2 Plugin
  useEffect(() => {
    let isMounted = true;

    const setupShortcut = async () => {
      try {
        await unregisterAll();
        await register(hotkey, async (event: ShortcutEvent) => {
          if (!isMounted) return;

          if (event.state === "Pressed" && !recordingRef.current) {
            if (!apiKeyRef.current) {
              setStatusBanner("Please add your Groq API key in Settings first.");
              setActiveTab("settings");
              return;
            }
            recordingRef.current = true;
            setIsRecording(true);
            await emit("hud-state", { mode: "recording" });
            try {
              await invoke("start_recording");
            } catch (err) {
              console.error("Audio capture start error:", err);
              recordingRef.current = false;
              setIsRecording(false);
            }
          } else if (event.state === "Released" && recordingRef.current) {
            recordingRef.current = false;
            setIsRecording(false);
            await emit("hud-state", { mode: "processing" });

            try {
              await invoke<DictationEntry>("stop_and_transcribe", {
                apiKey: apiKeyRef.current,
              });
              await fetchHistory();
              setStatusBanner(null);
            } catch (err) {
              console.error("Transcription failed:", err);
              await emit("hud-state", { mode: "error" });
              setStatusBanner(String(err));
            }
          }
        });
      } catch (err) {
        console.warn("Global shortcut registration note:", err);
      }
    };

    setupShortcut();
    return () => {
      isMounted = false;
      unregisterAll().catch(() => {});
    };
  }, [hotkey]);

  // Manual toggle button for testing inside the Dashboard UI
  const toggleManualDictation = async () => {
    if (!apiKey) {
      setStatusBanner("Enter your Groq API key in the Settings tab first.");
      setActiveTab("settings");
      return;
    }

    if (!isRecording) {
      recordingRef.current = true;
      setIsRecording(true);
      await emit("hud-state", { mode: "recording" });
      await invoke("start_recording");
    } else {
      recordingRef.current = false;
      setIsRecording(false);
      await emit("hud-state", { mode: "processing" });
      try {
        await invoke("stop_and_transcribe", { apiKey });
        await fetchHistory();
      } catch (err) {
        setStatusBanner(String(err));
      }
    }
  };

  const handleGenerateDigest = async () => {
    if (!apiKey || entries.length === 0) return;
    setIsGeneratingDigest(true);
    try {
      const joinedLogs = entries
        .slice(0, 40)
        .map(
          (e) =>
            `[${e.timestamp}] (App: ${e.active_app}): ${e.polished_text}`
        )
        .join("\n");

      const summary = await invoke<string>("generate_daily_digest_cmd", {
        apiKey,
        joinedLogs,
      });
      setDigestMarkdown(summary);
    } catch (err) {
      setDigestMarkdown(`Failed to generate summary: ${String(err)}`);
    } finally {
      setIsGeneratingDigest(false);
    }
  };

  const copyText = (id: number, text: string) => {
    navigator.clipboard.writeText(text);
    setCopiedId(id);
    setTimeout(() => setCopiedId(null), 1500);
  };

  const totalWords = entries.reduce((acc, item) => acc + item.word_count, 0);
  const totalSeconds = entries.reduce((acc, item) => acc + item.duration_sec, 0);
  const avgWpm =
    totalSeconds > 5 ? Math.round((totalWords / totalSeconds) * 60) : 0;

  return (
    <div className="min-h-screen bg-zinc-950 text-zinc-100 flex flex-col font-sans">
      {/* Top Header */}
      <header className="border-b border-zinc-800/80 px-6 py-4 flex items-center justify-between bg-zinc-900/40">
        <div className="flex items-center gap-3">
          <div className="h-9 w-9 rounded-xl bg-emerald-500/10 border border-emerald-500/30 flex items-center justify-center">
            <Mic className="w-5 h-5 text-emerald-400" />
          </div>
          <div>
            <h1 className="text-base font-semibold tracking-tight flex items-center gap-2">
              Suki for Windows
              <span className="text-[11px] font-mono px-2 py-0.5 rounded-full bg-zinc-800 text-zinc-400">
                BYOK v0.1
              </span>
            </h1>
            <p className="text-xs text-zinc-400">
              Hold <kbd className="px-1.5 py-0.5 bg-zinc-800 rounded text-zinc-200 font-mono">{hotkey}</kbd> anywhere to dictate
            </p>
          </div>
        </div>

        <button
          onClick={toggleManualDictation}
          className={`px-4 py-2 rounded-lg text-xs font-medium flex items-center gap-2 transition ${
            isRecording
              ? "bg-red-500/20 text-red-300 border border-red-500/40 animate-pulse"
              : "bg-emerald-500 text-zinc-950 hover:bg-emerald-400"
          }`}
        >
          <Mic className="w-3.5 h-3.5" />
          {isRecording ? "Stop & Transcribe" : "Test Dictation"}
        </button>
      </header>

      {statusBanner && (
        <div className="bg-amber-500/10 border-b border-amber-500/30 px-6 py-2 text-xs text-amber-300 flex justify-between items-center">
          <span>{statusBanner}</span>
          <button onClick={() => setStatusBanner(null)} className="underline">
            Dismiss
          </button>
        </div>
      )}

      {/* Metrics Bar */}
      <div className="grid grid-cols-3 gap-4 px-6 py-4 border-b border-zinc-900 bg-zinc-900/20">
        <div className="bg-zinc-900/60 border border-zinc-800/70 rounded-xl p-3 flex items-center gap-3">
          <Zap className="w-5 h-5 text-emerald-400" />
          <div>
            <div className="text-xs text-zinc-400">Total Words Dictated</div>
            <div className="text-lg font-semibold">{totalWords.toLocaleString()}</div>
          </div>
        </div>
        <div className="bg-zinc-900/60 border border-zinc-800/70 rounded-xl p-3 flex items-center gap-3">
          <Clock className="w-5 h-5 text-sky-400" />
          <div>
            <div className="text-xs text-zinc-400">Speaking Speed</div>
            <div className="text-lg font-semibold">{avgWpm} WPM</div>
          </div>
        </div>
        <div className="bg-zinc-900/60 border border-zinc-800/70 rounded-xl p-3 flex items-center gap-3">
          <Terminal className="w-5 h-5 text-purple-400" />
          <div>
            <div className="text-xs text-zinc-400">Saved Dictations</div>
            <div className="text-lg font-semibold">{entries.length}</div>
          </div>
        </div>
      </div>

      {/* Navigation Tabs */}
      <div className="flex gap-2 px-6 pt-3 border-b border-zinc-800/80">
        {[
          { id: "history", label: "Dictation History", icon: History },
          { id: "insights", label: "AI Daily Digest", icon: Sparkles },
          { id: "settings", label: "BYOK Settings", icon: Settings },
        ].map((tab) => {
          const Icon = tab.icon;
          const active = activeTab === tab.id;
          return (
            <button
              key={tab.id}
              onClick={() => setActiveTab(tab.id as typeof activeTab)}
              className={`flex items-center gap-2 px-4 py-2.5 text-xs font-medium border-b-2 transition ${
                active
                  ? "border-emerald-400 text-emerald-400"
                  : "border-transparent text-zinc-400 hover:text-zinc-200"
              }`}
            >
              <Icon className="w-3.5 h-3.5" />
              {tab.label}
            </button>
          );
        })}
      </div>

      {/* Main Content Area */}
      <main className="flex-1 overflow-y-auto p-6">
        {activeTab === "history" && (
          <div className="space-y-3">
            {entries.length === 0 ? (
              <div className="text-center py-12 text-zinc-500 text-sm">
                No dictations recorded yet. Hold <kbd className="px-1.5 py-0.5 bg-zinc-900 rounded border border-zinc-800">{hotkey}</kbd> in any app and speak!
              </div>
            ) : (
              entries.map((item) => (
                <div
                  key={item.id}
                  className="bg-zinc-900/50 border border-zinc-800/80 rounded-xl p-4 hover:border-zinc-700 transition"
                >
                  <div className="flex items-center justify-between text-xs text-zinc-400 mb-2">
                    <span className="font-mono bg-zinc-800/80 text-zinc-300 px-2 py-0.5 rounded">
                      {item.active_app || "Desktop"}
                    </span>
                    <div className="flex items-center gap-3">
                      <span>{item.word_count} words • {item.duration_sec.toFixed(1)}s</span>
                      <span>{item.timestamp}</span>
                      <button
                        onClick={() => copyText(item.id, item.polished_text)}
                        className="text-zinc-300 hover:text-white flex items-center gap-1"
                      >
                        {copiedId === item.id ? (
                          <Check className="w-3.5 h-3.5 text-emerald-400" />
                        ) : (
                          <Copy className="w-3.5 h-3.5" />
                        )}
                      </button>
                    </div>
                  </div>
                  <p className="text-sm text-zinc-100 leading-relaxed">
                    {item.polished_text}
                  </p>
                  {item.raw_text !== item.polished_text && (
                    <p className="text-xs text-zinc-500 mt-2 italic">
                      Raw Whisper: "{item.raw_text}"
                    </p>
                  )}
                </div>
              ))
            )}
          </div>
        )}

        {activeTab === "insights" && (
          <div className="space-y-4">
            <div className="flex items-center justify-between bg-zinc-900/60 border border-zinc-800 rounded-xl p-4">
              <div>
                <h2 className="text-sm font-semibold">Voice Activity Digest</h2>
                <p className="text-xs text-zinc-400">
                  Uses Llama 3.3 70B to extract action items, topics, and work logs from your recent dictations.
                </p>
              </div>
              <button
                onClick={handleGenerateDigest}
                disabled={isGeneratingDigest || entries.length === 0}
                className="px-4 py-2 rounded-lg bg-emerald-500 text-zinc-950 font-medium text-xs hover:bg-emerald-400 disabled:opacity-40"
              >
                {isGeneratingDigest ? "Analyzing..." : "Generate Daily Digest"}
              </button>
            </div>

            {digestMarkdown && (
              <div className="bg-zinc-900/40 border border-zinc-800 rounded-xl p-5 whitespace-pre-wrap text-sm text-zinc-200 leading-relaxed">
                {digestMarkdown}
              </div>
            )}
          </div>
        )}

        {activeTab === "settings" && (
          <div className="max-w-lg space-y-5">
            <div>
              <label className="block text-xs font-medium text-zinc-300 mb-1.5 flex items-center gap-1.5">
                <KeyRound className="w-3.5 h-3.5 text-emerald-400" />
                Groq API Key (BYOK)
              </label>
              <input
                type="password"
                value={apiKey}
                onChange={(e) => {
                  setApiKey(e.target.value);
                  localStorage.setItem("suki_groq_api_key", e.target.value);
                }}
                placeholder="gsk_..."
                className="w-full bg-zinc-900 border border-zinc-800 rounded-lg px-3 py-2 text-sm text-zinc-100 focus:outline-none focus:border-emerald-500"
              />
              <p className="text-[11px] text-zinc-500 mt-1">
                Stored locally on your device. Powers Whisper-Large-V3-Turbo & Llama-3.3-70B.
              </p>
            </div>

            <div>
              <label className="block text-xs font-medium text-zinc-300 mb-1.5">
                Global Push-to-Talk Shortcut
              </label>
              <input
                type="text"
                value={hotkey}
                onChange={(e) => {
                  setHotkey(e.target.value);
                  localStorage.setItem("suki_hotkey", e.target.value);
                }}
                className="w-full bg-zinc-900 border border-zinc-800 rounded-lg px-3 py-2 text-sm font-mono text-zinc-100 focus:outline-none focus:border-emerald-500"
              />
              <p className="text-[11px] text-zinc-500 mt-1">
                Examples: <code>CommandOrControl+Shift+Space</code> or <code>Alt+Space</code>
              </p>
            </div>
          </div>
        )}
      </main>
    </div>
  );
}
```

## 4. Two Final Tauri v2 Glue Pieces (Don't Skip!)

In Tauri v2, plugins like `global-shortcut` and cross-window events require explicit capability permissions in `src-tauri/capabilities/default.json`. Without this file, `register()` will silently fail.

### 1. Create `src-tauri/capabilities/default.json`
```json
{
  "$schema": "../gen/schemas/desktop-schema.json",
  "identifier": "default",
  "description": "Permissions for Suki main and hud windows",
  "windows": ["main", "hud"],
  "permissions": [
    "core:default",
    "core:event:default",
    "core:window:default",
    "global-shortcut:allow-register",
    "global-shortcut:allow-unregister",
    "global-shortcut:allow-unregister-all",
    "global-shortcut:allow-is-registered"
  ]
}
```

### 2. Expose `generate_daily_digest_cmd` and the Shortcut Plugin in `src-tauri/src/main.rs`
Add this wrapper command in `main.rs` and include `.plugin(tauri_plugin_global_shortcut::Builder::new().build())` in your builder chain:

```rust
#[tauri::command]
async fn generate_daily_digest_cmd(api_key: String, joined_logs: String) -> Result<String, String> {
    groq::generate_daily_insights(&api_key, &joined_logs).await
}
```
And inside `main()`'s `tauri::Builder::default()`:
```rust
.plugin(tauri_plugin_global_shortcut::Builder::new().build())
.invoke_handler(tauri::generate_handler![
    get_active_window_context,
    start_recording,
    stop_and_transcribe,
    generate_daily_digest_cmd,
    db::get_recent_dictations,
    db::insert_dictation
])
```
