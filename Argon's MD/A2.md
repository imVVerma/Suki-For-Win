# gemini response

The complete, production-ready `audio.rs` and `groq.rs` modules, along with the **zero-latency clipboard injector** (`injector.rs`), form the core backend.

They are built around one strict rule: **nothing touches the disk or spawns a shell process during a dictation cycle**. The audio is captured straight into RAM, downsampled to a compact 16kHz mono WAV byte vector (so your payload is ~5x smaller than 48kHz stereo, making uploads instantaneous), sent to Groq, and pasted directly into your active window.

## 1. Zero-Disk In-Memory Audio Engine (`src-tauri/src/audio.rs`)

One tricky quirk in Rust with `cpal` is that `cpal::Stream` is not `Send` on Windows WASAPI, meaning you can't just store the `Stream` directly inside a Tauri global `Mutex`. To solve this cleanly without ever freezing the UI thread, we spawn a dedicated audio worker thread that listens for `Start` and `Stop` signals over a channel and returns a finalized `Vec<u8>` WAV file in memory.

```rust
use cpal::traits::{DeviceTrait, HostTrait, StreamTrait};
use hound::{SampleFormat, WavSpec, WavWriter};
use std::io::Cursor;
use std::sync::{mpsc, Arc, Mutex};
use std::thread;
use std::time::Instant;

pub enum AudioCommand {
    Start,
    Stop(mpsc::Sender<(Vec<u8>, f64)>), // Returns (WAV bytes in RAM, duration_sec)
}

pub struct AudioRecorder {
    tx: Mutex<mpsc::Sender<AudioCommand>>,
}

impl AudioRecorder {
    pub fn new() -> Self {
        let (tx, rx) = mpsc::channel::<AudioCommand>();

        // Dedicated background thread so cpal::Stream stays on its owning thread
        thread::spawn(move || {
            let host = cpal::default_host();
            let mut active_stream: Option<cpal::Stream> = None;
            let samples_buffer: Arc<Mutex<Vec<f32>>> = Arc::new(Mutex::new(Vec::new()));
            let mut sample_rate: u32 = 48000;
            let mut channels: u16 = 1;
            let mut start_time = Instant::now();

            while let Ok(cmd) = rx.recv() {
                match cmd {
                    AudioCommand::Start => {
                        samples_buffer.lock().unwrap().clear();
                        start_time = Instant::now();

                        if let Some(device) = host.default_input_device() {
                            if let Ok(config) = device.default_input_config() {
                                sample_rate = config.sample_rate().0;
                                channels = config.channels();
                                let buf_clone = Arc::clone(&samples_buffer);

                                let stream_res = match config.sample_format() {
                                    cpal::SampleFormat::F32 => device.build_input_stream(
                                        &config.into(),
                                        move |data: &[f32], _: &_| {
                                            if let Ok(mut buf) = buf_clone.lock() {
                                                buf.extend_from_slice(data);
                                            }
                                        },
                                        |err| eprintln!("Audio stream error: {}", err),
                                        None,
                                    ),
                                    cpal::SampleFormat::I16 => device.build_input_stream(
                                        &config.into(),
                                        move |data: &[i16], _: &_| {
                                            if let Ok(mut buf) = buf_clone.lock() {
                                                for &s in data {
                                                    buf.push(s as f32 / i16::MAX as f32);
                                                }
                                            }
                                        },
                                        |err| eprintln!("Audio stream error: {}", err),
                                        None,
                                    ),
                                    _ => continue,
                                };

                                if let Ok(stream) = stream_res {
                                    let _ = stream.play();
                                    active_stream = Some(stream);
                                }
                            }
                        }
                    }
                    AudioCommand::Stop(reply_tx) => {
                        // Drop the stream immediately to stop recording
                        active_stream = None;
                        let duration_sec = start_time.elapsed().as_secs_f64();

                        let raw_samples = samples_buffer.lock().unwrap().clone();
                        let wav_bytes = encode_to_16khz_mono_wav(&raw_samples, sample_rate, channels);
                        let _ = reply_tx.send((wav_bytes, duration_sec));
                    }
                }
            }
        });

        Self {
            tx: Mutex::new(tx),
        }
    }

    pub fn start(&self) -> Result<(), String> {
        self.tx
            .lock()
            .map_err(|e| e.to_string())?
            .send(AudioCommand::Start)
            .map_err(|e| e.to_string())
    }

    pub fn stop(&self) -> Result<(Vec<u8>, f64), String> {
        let (reply_tx, reply_rx) = mpsc::channel();
        self.tx
            .lock()
            .map_err(|e| e.to_string())?
            .send(AudioCommand::Stop(reply_tx))
            .map_err(|e| e.to_string())?;

        reply_rx.recv().map_err(|e| e.to_string())
    }
}

/// Mixes multi-channel audio down to Mono and downsamples to 16kHz PCM WAV in RAM
fn encode_to_16khz_mono_wav(samples: &[f32], src_rate: u32, channels: u16) -> Vec<u8> {
    let target_rate = 16000u32;
    let ch = channels.max(1) as usize;

    // 1. Convert interleaved stereo/multi-channel to mono
    let mono: Vec<f32> = samples
        .chunks(ch)
        .map(|frame| frame.iter().sum::<f32>() / (ch as f32))
        .collect();

    // 2. Fast linear decimation to 16kHz (Whisper's native sample rate)
    let ratio = src_rate as f64 / target_rate as f64;
    let out_len = ((mono.len() as f64) / ratio) as usize;

    let spec = WavSpec {
        channels: 1,
        sample_rate: target_rate,
        bits_per_sample: 16,
        sample_format: SampleFormat::Int,
    };

    let mut cursor = Cursor::new(Vec::with_capacity(out_len * 2 + 44));
    {
        let mut writer = WavWriter::new(&mut cursor, spec).expect("Failed to create in-memory WAV writer");
        for i in 0..out_len {
            let src_idx = ((i as f64) * ratio) as usize;
            if let Some(&sample) = mono.get(src_idx) {
                let clamped = sample.clamp(-1.0, 1.0);
                let pcm_i16 = (clamped * (i16::MAX as f32)) as i16;
                let _ = writer.write_sample(pcm_i16);
            }
        }
        let _ = writer.finalize();
    }

    cursor.into_inner()
}
```

## 2. Groq Whisper + Context-Aware Llama Pipeline (`src-tauri/src/groq.rs`)

This module takes the `Vec<u8>` WAV bytes directly from `audio.rs`, uploads them as an in-memory multipart form (`recording.wav`) to Groq's `whisper-large-v3-turbo`, and then runs a super-fast formatting pass on `llama-3.3-70b-versatile` tailored to whatever app you have focused.

```rust
use reqwest::multipart::{Form, Part};
use serde::{Deserialize, Serialize};
use serde_json::json;

#[derive(Deserialize)]
struct WhisperResponse {
    text: String,
}

#[derive(Deserialize)]
struct ChatCompletionResponse {
    choices: Vec<ChatChoice>,
}

#[derive(Deserialize)]
struct ChatChoice {
    message: ChatMessage,
}

#[derive(Deserialize, Serialize)]
struct ChatMessage {
    role: String,
    content: String,
}

/// Step 1: Send in-memory WAV bytes to Groq Whisper-Large-V3-Turbo
pub async fn transcribe_wav_bytes(api_key: &str, wav_bytes: Vec<u8>) -> Result<String, String> {
    if wav_bytes.len() <= 44 {
        return Err("Recording too short or empty".to_string());
    }

    let client = reqwest::Client::new();
    let audio_part = Part::bytes(wav_bytes)
        .file_name("dictation.wav")
        .mime_str("audio/wav")
        .map_err(|e| e.to_string())?;

    let form = Form::new()
        .part("file", audio_part)
        .text("model", "whisper-large-v3-turbo")
        .text("temperature", "0.0")
        .text("response_format", "json");

    let res = client
        .post("https://api.groq.com/openai/v1/audio/transcriptions")
        .bearer_auth(api_key)
        .multipart(form)
        .send()
        .await
        .map_err(|e| format!("Network error contacting Groq Whisper: {}", e))?;

    if !res.status().is_success() {
        let err_body = res.text().await.unwrap_or_default();
        return Err(format!("Groq Whisper API Error: {}", err_body));
    }

    let parsed: WhisperResponse = res.json().await.map_err(|e| e.to_string())?;
    Ok(parsed.text.trim().to_string())
}

/// Step 2: Polish transcript with Llama 3.3 70B using the active Windows app title as context
pub async fn polish_transcript(
    api_key: &str,
    raw_text: &str,
    active_window_title: &str,
) -> Result<String, String> {
    if raw_text.is_empty() {
        return Ok(String::new());
    }

    let system_prompt = format!(
        "You are an invisible dictation cleanup engine (like Wispr Flow / Suki). \
         The user is currently typing inside the application window: '{}'. \
         Rules:\n\
         1. Remove spoken disfluencies, stutters, and filler words ('um', 'uh', 'like', 'you know').\n\
         2. If the user corrects themselves mid-sentence ('meet at 4 no wait 5 PM'), output only the final intended meaning ('meet at 5 PM').\n\
         3. Adapt formatting subtly to the active window (e.g., if in Visual Studio Code or terminal, preserve technical terms, variable casing, and CLI flags accurately; if in email/chat, use natural punctuation).\n\
         4. Output ONLY the final cleaned text. Never add quotes, preambles, or explanations.",
        active_window_title
    );

    let payload = json!({
        "model": "llama-3.3-70b-versatile",
        "temperature": 0.1,
        "messages": [
            { "role": "system", "content": system_prompt },
            { "role": "user", "content": raw_text }
        ]
    });

    let client = reqwest::Client::new();
    let res = client
        .post("https://api.groq.com/openai/v1/chat/completions")
        .bearer_auth(api_key)
        .json(&payload)
        .send()
        .await
        .map_err(|e| format!("Network error contacting Groq LLM: {}", e))?;

    if !res.status().is_success() {
        // Fallback gracefully to raw Whisper text if LLM rate-limits
        return Ok(raw_text.to_string());
    }

    let parsed: ChatCompletionResponse = res.json().await.map_err(|e| e.to_string())?;
    let polished = parsed
        .choices
        .first()
        .map(|c| c.message.content.trim().to_string())
        .unwrap_or_else(|| raw_text.to_string());

    Ok(polished)
}

/// Bonus Step 3: Generate an AI Daily Digest from SQLite logs
pub async fn generate_daily_insights(api_key: &str, joined_logs: &str) -> Result<String, String> {
```

## 3. Non-Destructive Clipboard Text Injector (`src-tauri/src/injector.rs`)

Instead of typing character-by-character (which takes 2–3 seconds for a paragraph and triggers auto-complete popups in IDEs), Suki backs up your current clipboard, swaps in the polished text, fires a synthetic `Ctrl+V`, and restores your original clipboard 60ms later.

```rust
use arboard::Clipboard;
use enigo::{Direction, Enigo, Key, Keyboard, Settings};
use std::thread;
use std::time::Duration;

pub fn inject_text_at_cursor(text: &str) -> Result<(), String> {
    if text.is_empty() {
        return Ok(());
    }

    let mut clipboard = Clipboard::new().map_err(|e| e.to_string())?;

    // 1. Back up the user's current clipboard text (if any)
    let previous_clipboard = clipboard.get_text().ok();

    // 2. Place polished dictation into clipboard
    clipboard.set_text(text.to_string()).map_err(|e| e.to_string())?;

    // Small settle delay for Windows clipboard manager
    thread::sleep(Duration::from_millis(25));

    // 3. Synthesize Ctrl + V into the currently focused window
    let mut enigo = Enigo::new(&Settings::default()).map_err(|e| e.to_string())?;
    let _ = enigo.key(Key::Control, Direction::Press);
    let _ = enigo.key(Key::Unicode('v'), Direction::Click);
    let _ = enigo.key(Key::Control, Direction::Release);

    // 4. Wait 60ms for target app (VS Code, Arc, etc.) to process paste, then restore old clipboard
    thread::sleep(Duration::from_millis(60));
    if let Some(old_text) = previous_clipboard {
        let _ = clipboard.set_text(old_text);
    }

    Ok(())
}
```

## 4. Wiring the Commands Together in `src-tauri/src/main.rs`

Add `mod audio; mod groq; mod injector;` at the top of `main.rs`, register `app.manage(audio::AudioRecorder::new());` inside `.setup()`, and expose these two commands to your React frontend or Global Shortcut handler:

```rust
#[tauri::command]
fn start_recording(
    app: tauri::AppHandle,
    recorder: tauri::State<'_, audio::AudioRecorder>,
) -> Result<(), String> {
    if let Some(hud) = app.get_webview_window("hud") {
        let _ = hud.show(); // Shows without stealing focus thanks to WS_EX_NOACTIVATE!
    }
    recorder.start()
}

#[tauri::command]
async fn stop_and_transcribe(
    app: tauri::AppHandle,
    recorder: tauri::State<'_, audio::AudioRecorder>,
    db: tauri::State<'_, std::sync::Mutex<rusqlite::Connection>>,
    api_key: String,
) -> Result<db::DictationEntry, String> {
    // 1. Grab the active window title BEFORE anything changes
    let active_app = get_active_window_context();

    // 2. Stop audio capture and get in-memory 16kHz WAV bytes
    let (wav_bytes, duration_sec) = recorder.stop()?;

    // 3. Transcribe with Whisper-Large-V3-Turbo & Polish with Llama-3.3-70B
    let raw_text = groq::transcribe_wav_bytes(&api_key, wav_bytes).await?;
    let polished_text = groq::polish_transcript(&api_key, &raw_text, &active_app).await?;

    // 4. Hide HUD and inject text directly into the active window
    if let Some(hud) = app.get_webview_window("hud") {
        let _ = hud.hide();
    }
    injector::inject_text_at_cursor(&polished_text)?;

    // 5. Save to %APPDATA% SQLite history
    db::insert_dictation(
        db,
        raw_text.clone(),
        polished_text.clone(),
        active_app.clone(),
        duration_sec,
    )?;

    Ok(db::DictationEntry {
        id: 0,
        timestamp: "Just now".to_string(),
        raw_text,
        polished_text,
        active_app,
        duration_sec,
        word_count: 0,
    })
}
```
